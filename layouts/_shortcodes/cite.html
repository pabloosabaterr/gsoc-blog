{{- /* Inline citation to an entry in the post's `references`. Usage: {{< cite 1 >}} (by number) or {{< cite matklad >}} (by id). */ -}} {{- $key := .Get 0 }} {{- $refs := .Page.Params.references }} {{- $n := 0 }}{{ $r := dict }} {{- range $i, $e := $refs }} {{- if or (eq (string (add $i 1)) (string $key)) (eq (string $e.id) (string $key)) }}{{ $n = add $i 1 }}{{ $r = $e }}{{ end }} {{- end }} {{- if $n -}} [{{ $n }}] {{- else }}{{ errorf "cite: no reference %q in %s" $key .Page.File.Path }}{{ end -}}
